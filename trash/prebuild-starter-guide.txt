        @Column(length = 500)
        private String evidencePointer;

    package com.verdict.service;

    import com.verdict.dto.AuditRequest;
    import com.verdict.dto.AuditResponse;
    import com.verdict.dto.SummaryResponse;
    import com.verdict.model.Audit;
    import com.verdict.model.Verdict;
    import com.verdict.repository.AuditRepository;
    import com.verdict.repository.VerdictRepository;
    import org.springframework.stereotype.Service;
    import org.springframework.transaction.annotation.Transactional;

    import java.util.*;

    @Service
    public class AuditService {

        private final AuditRepository auditRepo;
        private final VerdictRepository verdictRepo;
        private final LlmService llmService;

        public AuditService(AuditRepository auditRepo,
                            VerdictRepository verdictRepo,
                            LlmService llmService) {
            this.auditRepo = auditRepo;
            this.verdictRepo = verdictRepo;
            this.llmService = llmService;
        }

        @Transactional
        public AuditResponse runAudit(AuditRequest request, String sessionId) {
            // 1. Split claim text into sentences
            List<String> sentences = splitSentences(request.getClaimText());

            // 2. Send ALL sentences + evidence in ONE LLM call
            //    The LLM returns a JSON array with isClaim + status per sentence
            List<AuditResponse.VerdictItem> llmVerdicts = llmService.auditBatch(
                sentences, request.getEvidenceText()
            );

            // 3. Compute credibility score
            int totalClaims = 0;
            int supported = 0;
            int stretched = 0;
            int unsupported = 0;
            int notClaims = 0;

            for (AuditResponse.VerdictItem item : llmVerdicts) {
                if (item.isClaim()) {
                    totalClaims++;
                    switch (item.getStatus()) {
                        case "SUPPORTED"   -> supported++;
                        case "STRETCHED"   -> stretched++;
                        case "UNSUPPORTED" -> unsupported++;
                    }
                } else {
                    notClaims++;
                }
            }

            int score = totalClaims > 0
                ? (int) Math.round((supported * 1.0 + stretched * 0.4) / totalClaims * 100)
                : 100;

            // 4. Save to database
            Audit audit = new Audit();
            audit.setClaimTitle(request.getClaimTitle());
            audit.setEvidenceTitle(request.getEvidenceTitle());
            audit.setClaimText(request.getClaimText());
            audit.setEvidenceText(request.getEvidenceText());
            audit.setCredibilityScore(score);
            audit.setTotalSentences(sentences.size());
            audit.setSupportedCount(supported);
            audit.setStretchedCount(stretched);
            audit.setUnsupportedCount(unsupported);
            audit.setNotClaimCount(notClaims);
            audit.setSessionId(sessionId);
            audit = auditRepo.save(audit);

            final Long auditId = audit.getId();
            for (AuditResponse.VerdictItem item : llmVerdicts) {
                if (item.isClaim()) {
                    Verdict v = new Verdict();
                    v.setAudit(audit);
                    v.setSentenceIndex(item.getSentenceIndex());
                    v.setSentenceText(item.getSentence());
                    v.setClaim(true);
                    v.setStatus(item.getStatus());
                    v.setExplanation(item.getExplanation());
                    v.setEvidencePointer(item.getEvidencePointer());
                    verdictRepo.save(v);
                } else {
                    Verdict v = new Verdict();
                    v.setAudit(audit);
                    v.setSentenceIndex(item.getSentenceIndex());
                    v.setSentenceText(item.getSentence());
                    v.setClaim(false);
                    v.setStatus(null);
                    v.setExplanation(null);
                    v.setEvidencePointer(null);
                    verdictRepo.save(v);
                }
            }

            // 5. Build response
            AuditResponse response = new AuditResponse();
            response.setAuditId(auditId);
            response.setCredibilityScore(score);
            response.setVerdicts(llmVerdicts);
            response.setSummary(supported + " SUPPORTED, " + stretched + " STRETCHED, "
                + unsupported + " UNSUPPORTED, " + notClaims + " NOT-A-CLAIM");
            response.setCreatedAt(audit.getCreatedAt());
            return response;
        }

        public AuditResponse getAudit(Long id) {
            Audit audit = auditRepo.findById(id).orElseThrow();
            List<Verdict> verdicts = verdictRepo.findByAuditIdOrderBySentenceIndexAsc(id);

            AuditResponse response = new AuditResponse();
            response.setAuditId(audit.getId());
            response.setCredibilityScore(audit.getCredibilityScore());
            response.setCreatedAt(audit.getCreatedAt());
            response.setSummary(audit.getSupportedCount() + " SUPPORTED, "
                + audit.getStretchedCount() + " STRETCHED, "
                + audit.getUnsupportedCount() + " UNSUPPORTED");

            List<AuditResponse.VerdictItem> items = verdicts.stream()
                .map(v -> {
                    AuditResponse.VerdictItem item = new AuditResponse.VerdictItem();
                    item.setSentenceIndex(v.getSentenceIndex());
                    item.setSentence(v.getSentenceText());
                    item.setClaim(v.isClaim());
                    item.setStatus(v.getStatus());
                    item.setExplanation(v.getExplanation());
                    item.setEvidencePointer(v.getEvidencePointer());
                    return item;
                })
                .toList();
            response.setVerdicts(items);
            return response;
        }

        public SummaryResponse getSummary(Long id) {
            Audit audit = auditRepo.findById(id).orElseThrow();
            List<Verdict> verdicts = verdictRepo.findByAuditIdOrderBySentenceIndexAsc(id);

            SummaryResponse sum = new SummaryResponse();
            sum.setAuditId(audit.getId());
            sum.setClaimTitle(audit.getClaimTitle());
            sum.setEvidenceTitle(audit.getEvidenceTitle());
            sum.setCredibilityScore(audit.getCredibilityScore());
            sum.setTotalSentences(audit.getTotalSentences());
            sum.setSupportedCount(audit.getSupportedCount());
            sum.setStretchedCount(audit.getStretchedCount());
            sum.setUnsupportedCount(audit.getUnsupportedCount());
            sum.setNotClaimCount(audit.getNotClaimCount());

            // Top 3 most damaging = UNSUPPORTED first, then STRETCHED
            List<SummaryResponse.DamageItem> top = verdicts.stream()
                .filter(v -> v.isClaim() && !"SUPPORTED".equals(v.getStatus()))
                .sorted(Comparator.comparingInt(v ->
                    "UNSUPPORTED".equals(v.getStatus()) ? 0 : 1))
                .limit(3)
                .map(v -> {
                    SummaryResponse.DamageItem d = new SummaryResponse.DamageItem();
                    d.setSentence(v.getSentenceText());
                    d.setExplanation(v.getExplanation());
                    return d;
                })
                .toList();
            sum.setTopUnsupported(top);
            return sum;
        }

        public List<Audit> getHistory(String sessionId) {
            return auditRepo.findBySessionIdOrderByCreatedAtDesc(sessionId);
        }

        private List<String> splitSentences(String text) {
            // Simple split by period + newline. Good enough for CVs and reports.
            if (text == null || text.isBlank()) return List.of();
            return Arrays.stream(text.split("(?<=[.!?])\\s*|\\n\\s*"))
                .map(String::trim)
                .filter(s -> !s.isEmpty())
                .toList();
        }
    }

    package com.verdict.controller;

    import com.verdict.dto.AuditRequest;
    import com.verdict.dto.AuditResponse;
    import com.verdict.dto.SummaryResponse;
    import com.verdict.model.Audit;
    import com.verdict.service.AuditService;
    import jakarta.servlet.http.HttpSession;
    import jakarta.validation.Valid;
    import org.springframework.http.ResponseEntity;
    import org.springframework.web.bind.annotation.*;

    import java.util.List;

    @RestController
    @RequestMapping("/api")
    public class AuditController {

        private final AuditService auditService;

        public AuditController(AuditService auditService) {
            this.auditService = auditService;
        }

        @PostMapping("/audit/run")
        public ResponseEntity<AuditResponse> runAudit(
                @Valid @RequestBody AuditRequest request,
                HttpSession session) {
            String sessionId = session.getId();
            AuditResponse response = auditService.runAudit(request, sessionId);
            return ResponseEntity.ok(response);
        }

        @GetMapping("/audit/{id}")
        public ResponseEntity<AuditResponse> getAudit(@PathVariable Long id) {
            return ResponseEntity.ok(auditService.getAudit(id));
        }

        @GetMapping("/audit/{id}/summary")
        public ResponseEntity<SummaryResponse> getSummary(@PathVariable Long id) {
            return ResponseEntity.ok(auditService.getSummary(id));
        }

        @GetMapping("/audit/history")
        public ResponseEntity<List<Audit>> getHistory(HttpSession session) {
            return ResponseEntity.ok(auditService.getHistory(session.getId()));
        }

        @GetMapping("/admin/status")
        public ResponseEntity<String> health() {
            return ResponseEntity.ok("VERDICT running.");
        }
    }

    // ─── VERDICT Frontend ───

    let currentAuditId = null;
    let currentVerdicts = [];
    let currentFilter = 'all';

    // ─── Page Navigation ───
    function showPage(page) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.getElementById('page-' + page).classList.add('active');
        document.querySelectorAll('.nav-links a').forEach(a => a.classList.remove('active'));
        if (page === 'intake') {
            document.querySelector('.nav-links a[href="#"]').classList.add('active');
        } else if (page === 'history') {
            document.querySelectorAll('.nav-links a')[1].classList.add('active');
            loadHistory();
        }
    }

    // ─── File Upload ───
    function loadFile(target) {
        const input = document.getElementById(target + '-file');
        const textarea = document.getElementById(target + '-text');
        if (input.files && input.files[0]) {
            const reader = new FileReader();
            reader.onload = function(e) {
                textarea.value = e.target.result;
            };
            reader.readAsText(input.files[0]);
        }
    }

    // ─── Run Audit ───
    async function runAudit() {
        const claimText = document.getElementById('claim-text').value.trim();
        const evidenceText = document.getElementById('evidence-text').value.trim();

        if (!claimText || !evidenceText) {
            document.getElementById('status-text').textContent = 'ERROR: BOTH FIELDS REQUIRED';
            return;
        }

        document.getElementById('status-text').textContent = '';
        document.getElementById('loading-overlay').classList.remove('hidden');

        try {
            const response = await fetch('/api/audit/run', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    claimTitle: 'Claim Document',
                    evidenceTitle: 'Evidence Document',
                    claimText: claimText,
                    evidenceText: evidenceText
                })
            });

            if (!response.ok) throw new Error('HTTP ' + response.status);

            const data = await response.json();
            currentAuditId = data.auditId;
            currentVerdicts = data.verdicts;
            currentFilter = 'all';

            renderVerdicts(data);
            showPage('verdict');

        } catch (err) {
            document.getElementById('status-text').textContent = 'ERROR: ' + err.message;
        } finally {
            document.getElementById('loading-overlay').classList.add('hidden');
        }
    }

    // ─── Render Verdicts ───
    function renderVerdicts(data) {
        document.getElementById('score-display').textContent = data.credibilityScore + '%';

        const list = document.getElementById('verdict-list');
        list.innerHTML = '';

        data.verdicts.forEach(item => {
            if (currentFilter === 'problems' && (!item.claim || item.status === 'SUPPORTED')) return;
            if (currentFilter === 'passed' && (!item.claim || item.status !== 'SUPPORTED')) return;

            const div = document.createElement('div');
            div.className = 'verdict-item ' + (item.claim ? item.status : 'NOT-A-CLAIM');

            let badgeText = item.claim ? item.status : 'NOT-A-CLAIM';
            div.innerHTML = `
                <span class="badge ${badgeText}">${badgeText}</span>
                <div class="sentence">${escapeHtml(item.sentence)}</div>
                ${item.explanation ? '<div class="explanation">' + escapeHtml(item.explanation) + '</div>' : ''}
                ${item.evidencePointer ? '<div class="pointer">→ ' + escapeHtml(item.evidencePointer) + '</div>' : ''}
            `;
            list.appendChild(div);
        });
    }

    // ─── Filter ───
    function setFilter(filter) {
        currentFilter = filter;
        document.querySelectorAll('.filter-btn').forEach(btn => {
            btn.classList.toggle('active', btn.dataset.filter === filter);
        });
        // Re-render from stored data
        const data = {
            credibilityScore: document.getElementById('score-display').textContent.replace('%', ''),
            verdicts: currentVerdicts
        };
        renderVerdicts(data);
    }

    // ─── View Summary ───
    async function viewSummary() {
        if (!currentAuditId) return;
        try {
            const response = await fetch('/api/audit/' + currentAuditId + '/summary');
            const data = await response.json();
            renderSummary(data);
            showPage('summary');

        }
    }

    function renderSummary(data) {
        const el = document.getElementById('summary-content');
        let html = '';
        html += '<div class="stat-row"><span>AUDIT ID</span><span>' + data.auditId + '</span></div>';
        html += '<div class="stat-row"><span>CREDIBILITY SCORE</span><span>' + data.credibilityScore + '%</span></div>';
        html += '<div class="stat-row"><span>TOTAL SENTENCES</span><span>' + data.totalSentences + '</span></div>';
        html += '<div class="stat-row"><span>SUPPORTED</span><span style="color:#2a6b3c">' + data.supportedCount + '</span></div>';
        html += '<div class="stat-row"><span>STRETCHED</span><span style="color:#b8860b">' + data.stretchedCount + '</span></div>';
        html += '<div class="stat-row"><span>UNSUPPORTED</span><span style="color:#8b2a2a">' + data.unsupportedCount + '</span></div>';
        html += '<div class="stat-row"><span>NOT-A-CLAIM</span><span style="color:#4a4a4a">' + data.notClaimCount + '</span></div>';

        if (data.topUnsupported && data.topUnsupported.length > 0) {
            html += '<h2 style="margin-top:20px">TOP UNSUPPORTED CLAIMS</h2>';
            data.topUnsupported.forEach(d => {
                html += '<div class="damage-item">';
                html += '<div class="damage-sentence">' + escapeHtml(d.sentence) + '</div>';
                html += '<div class="damage-explanation">' + escapeHtml(d.explanation) + '</div>';
                html += '</div>';
            });
        }

        el.innerHTML = html;
    }

    // ─── Load History ───
    async function loadHistory() {
        try {
            const response = await fetch('/api/audit/history');
            const data = await response.json();
            const el = document.getElementById('history-list');
            el.innerHTML = '';
            if (data.length === 0) {
                el.innerHTML = '<div style="color:#4a4a4a; text-align:center; padding:20px;">NO AUDITS RECORDED</div>';
                return;
            }
            data.forEach(audit => {
                const div = document.createElement('div');
                div.className = 'history-item';
                div.onclick = function() {
                    currentAuditId = audit.id;
                    fetch('/api/audit/' + audit.id)
                        .then(r => r.json())
                        .then(data => {
                            currentVerdicts = data.verdicts;
                            currentFilter = 'all';
                            renderVerdicts(data);
                            showPage('verdict');
                        });
                };
                div.innerHTML = `
                    <div>
                        <div class="h-title">${escapeHtml(audit.claimTitle || 'Untitled')}</div>
                        <div class="h-date">${audit.createdAt ? new Date(audit.createdAt).toLocaleString() : ''}</div>
                    </div>
                    <div class="h-score" style="color:${
                        audit.credibilityScore >= 70 ? '#2a6b3c' :
                        audit.credibilityScore >= 40 ? '#b8860b' : '#8b2a2a'
                    }">${audit.credibilityScore}%</div>
                `;
                el.appendChild(div);
            });
        } catch (err) {
            document.getElementById('history-list').innerHTML = '<div style="color:#8b2a2a">ERROR LOADING HISTORY</div>';
        }
    }

    // ─── Utility ───
    function escapeHtml(str) {
        if (!str) return '';
        return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;').replace(/'/g, '&#039;');
    }
